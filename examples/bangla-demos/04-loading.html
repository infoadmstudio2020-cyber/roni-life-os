async function withLoading(text, task) {
  let shown = false;
  try {
    await Native.call('ui.showLoading', { text, cancelable: false });
    shown = true;
  } catch (e) {}
  try { return await task(); }
  finally {
    if (shown) await Native.call('ui.hideLoading').catch(() => {});
  }
}
